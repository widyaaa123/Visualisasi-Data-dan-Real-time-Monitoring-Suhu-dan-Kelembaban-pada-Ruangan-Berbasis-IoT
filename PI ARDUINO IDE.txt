#include <ESP8266WiFi.h>
#include <PubSubClient.h>
#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include <DHT.h>

// LCD I2C
LiquidCrystal_I2C lcd(0x27, 16, 2);

// WiFi
const char *ssid = "VMIN9T";
const char *password = "azril123";

// MQTT Broker
const char *mqtt_broker = "192.168.150.236";
const char *topic = "suhu";
const int mqtt_port = 1883;

// DHT22
#define DHTPIN D5
#define DHTTYPE DHT22
DHT dht(DHTPIN, DHTTYPE);

WiFiClient espClient;
PubSubClient client(espClient);
unsigned long lastMsg = 0;

void setup() {
  Serial.begin(115200);

  lcd.begin();
  lcd.backlight();
  lcd.setCursor(0, 0);
  lcd.print("Connecting WiFi");

  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.println("Connecting to WiFi..");
  }
  Serial.println("Connected to the WiFi");
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("WiFi Connected");

  client.setServer(mqtt_broker, mqtt_port);
  client.setCallback(callback);

  while (!client.connected()) {
    String client_id = "esp8266-client-";
    client_id += String(WiFi.macAddress());
    Serial.printf("Connecting to MQTT as %s\n", client_id.c_str());

    if (client.connect(client_id.c_str())) {
      Serial.println("MQTT Connected");
      lcd.setCursor(0, 1);
      lcd.print("MQTT Connected");
    } else {
      Serial.print("MQTT Failed: ");
      Serial.println(client.state());
      lcd.setCursor(0, 1);
      lcd.print("Retrying MQTT..");
      delay(2000);
    }
  }

  client.subscribe(topic);
  dht.begin();
}

void callback(char *topic, byte *payload, unsigned int length) {
  Serial.print("Message arrived [");
  Serial.print(topic);
  Serial.print("]: ");

  String message = "";
  for (int i = 0; i < length; i++) {
    message += (char)payload[i];
  }
  Serial.println(message);

  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("From MQTT:");
  lcd.setCursor(0, 1);
  lcd.print(message.substring(0, 16));
}

void loop() {
  client.loop();

  unsigned long now = millis();
  if (now - lastMsg > 30000) { // Kirim tiap 30 detik
    lastMsg = now;

    float suhu = dht.readTemperature();
    float kelembaban = dht.readHumidity();

    if (isnan(suhu) || isnan(kelembaban)) {
      Serial.println("Failed to read from DHT sensor!");
      return;
    }

    // Gabungkan suhu & kelembapan jadi 1 string
    String payload = "Suhu: " + String(suhu, 1) + "C, H: " + String(kelembaban, 1) + "%";
    Serial.println("Publish: " + payload);

    client.publish(topic, payload.c_str());

    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print("Suhu: " + String(suhu, 1) + "C");
    lcd.setCursor(0, 1);
    lcd.print("H: " + String(kelembaban, 1) + "%");
  }
}
